HighLight CF =
VAR min_val =
    MINX (
        ALL ( Kalender[Måned], Kalender[MånedNr] ),
	[Salg]
    )
VAR max_val =
    MAXX (
        ALL ( Kalender[Måned], Kalender[MånedNr] ),
	[Salg]
    )
VAR color = 
    SWITCH (
        TRUE(),
	[Salg] = min_val, "bad",
	[Salg] = max_val, "good",
	"neutral"
    )
RETURN
    color

VAR BaselineVal = [Revenue PY]
VAR ValToCompare = [Revenue]

    IF (

	 " ▼"
    )

    FORMAT ( 
        DIVIDE (
	    ValToCompare - BaselineVal,
	    BaselineVal
	),
	" 0.00%"
