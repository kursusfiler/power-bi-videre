(#"Start Year" as number, #"End Year" as number) as table =>
let
	GenererDatoer = List.Dates(#date(#"Start Year", 1, 1), Duration.Days(Duration.From(#date(#"End Year", 12, 31) - 
	#date(#"Start Year" - 1, 12, 31))), 
	#duration( 1, 0, 0, 0) ),
	#"Konverter til tabel" = Table.TransformColumnTypes(Table.FromList(GenererDatoer, Splitter.SplitByNothing(), {"Date"}), {"Date", Date.Type}),
	#"Tilføj år" = Table.AddColumn(#"Konverter til tabel", "Year", each Date.Year([Date]), Int64.Type),
	#"Tilføj kvartal" = Table.AddColumn(#"Tilføj år", "Quarter", each "Qtr " & Text.From(Date.QuarterOfYear([Date])) , Text.Type),
	#"Tilføj månedsnr" = Table.AddColumn(#"Tilføj kvartal", "MonthOrder", each Date.ToText([Date], "MM"), Text.Type),
	#"Tilføj kort månedsnavn" = Table.AddColumn(#"Tilføj månedsnr", "Month Short", each Date.ToText([Date], "MMM", "en-US"), Text.Type),
	#"Tilføj månedsnavn" = Table.AddColumn(#"Tilføj kort månedsnavn", "Month", each Date.MonthName([Date], "en-US"), Text.Type)
in
	#"Tilføj månedsnavn"
