YOY% =
VAR CY = 
    SELECTEDMEASURE ()
VAR PY = 
    CALCULATE (
        SELECTEDMEASURE (),
        SAMEPERIODLASTYEAR ( 'Date'[Date] )
    )
VAR YOY = CY - PY
RETURN
    DIVIDE ( 
        YOY, 
        PY
    )
