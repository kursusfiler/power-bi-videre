Full Name =
VAR FirstName = Customer[First Name]
VAR MidName = Customer[Middle Name]
VAR LastName = Customer[Last Name]
VAR Result = 
    FirstName & " "
        & IF (
            MidName <> "",
            LEFT ( MidName, 1 ) & ". "
        ) & LastName
RETURN
	Result

Age Breakdown =
VAR CurrentAge = Customer[Current Age]
VAR Result =
    IF (
        CurrentAge >= 60,
        "60+",
        IF (
            CurrentAge >= 50,
            "50-59",
            IF (
                CurrentAge >= 40,
                "40-49",
                "18-39"
            )
        )
    )
RETURN
    Result
===============================
Age Breakdown =
VAR CurrentAge = Customer[Current Age]
VAR Result =
    SWITCH (
        TRUE (),
        CurrentAge >= 60, "60+",
        CurrentAge >= 50, "50-59",
        CurrentAge >= 40, "40-59",
        "18-39"
    )
RETURN
    Result

    [Total Sales],

Total Sales (All Countries) =
IF (
    ISBLANK ( [Total Sales] ),
    BLANK (),
    CALCULATE (
        [Total Sales],
        ALL ( 'Sales Territory'[Sales Territory Country] )
    )
)

    [Total Sales],
    [Total Sales (All Countries)]
)
